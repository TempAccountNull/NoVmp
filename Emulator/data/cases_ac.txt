# Alignment check (#AC, ledger U834): expected-value cases, Unicorn only, for what the i5-13600K
# cannot show (AVX-512, Intel APX, CPL0). Run with CR0.AM = 1:
#   emu-alltest --cases Emulator\data\cases_ac.txt --expect-only --avx512 --apx --cr0 0x40011
# SDM Vol3A 7.15 (event 17): CR0.AM = 1, EFLAGS.AC = 1 and CPL = 3 enable the check; a misaligned
# 2/4/8-byte data access faults with #AC(0) before the access. Vol2A 2.8: EVEX classes E2-E10 and
# K21 (KMOV) report #AC "for 2, 4, or 8 byte memory access"; E1/E4 full vectors, E11 (VCVTPS2PH,
# VCVTPH2PS) and E12 (VSIB) never do. APX PUSH2/POP2: #GP when RSP is not 16-byte aligned (APX
# spec 9.1.3), no #AC row; CCMPscc / CTESTscc (APX-EVEX-CCMP) and PUSHP/POPP report #AC.
#
# --- the enabling conditions: CPL0 never checks; AC = 0 never checks
mov eax, dword ptr [rsi+1] | rflags=0x40202 =>
mov eax, dword ptr [rsi+1] | cpl=3 rflags=0x202 =>
mov eax, dword ptr [rsi+1] | cpl=3 rflags=0x40202 => #AC
mov eax, dword ptr [rsi+4] | cpl=3 rflags=0x40202 =>
fxsave [rsi+2] | rflags=0x40202 => #GP
fxsave [rsi+2] | cpl=3 rflags=0x40202 => #AC
fxsave [rsi+4] | cpl=3 rflags=0x40202 => #GP
# --- AVX-512: scalar and broadcast operands are checked, full vectors are not

vmovsd xmm16, qword ptr [rsi+4] | cpl=3 rflags=0x40202 => #AC

vbroadcastss zmm0, dword ptr [rsi+2] | cpl=3 rflags=0x40202 => #AC

.byte 0x62, 0xe3, 0x7d, 0x08, 0x1d, 0x86, 0x04, 0, 0, 0, 0 | cpl=3 rflags=0x40202 =>!
vcvtph2ps zmm0, ymmword ptr [rsi+4] | cpl=3 rflags=0x40202 =>!

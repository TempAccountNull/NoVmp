# --- U772 exception error codes: an expected-value fault token may carry the error code,
# "#GP(0)" (Unicorn's UC_CTL_X86_EXCEPTION record, U770). SDM Vol1 3.3.7.1 (non-canonical: #SS(0)
# for a stack reference, else #GP(0)); Vol2A MOV / JMP / CALL protected-mode exceptions
# (#GP(selector) with the RPL bits cleared); Vol2 CLI / HLT / RDMSR / MOV CR at CPL3 #GP(0);
# Vol2 MOVDQA misaligned #GP(0).

mov rax, qword ptr [rcx] | rcx=0x8000000000000000 => #GP(0)

mov ax, 0x18; mov ds, ax | cpl=3 => rax=0x18 #GP(0x18)
mov ax, 0x43; mov ss, ax | cpl=3 => rax=0x43 #GP(0x40)
mov ax, 0x2b; mov ss, ax | cpl=3 => rax=0x2b
.byte 0xff, 0x2e | cpl=3 m+0x8000=000000004000 => #GP(0x40)
.byte 0xff, 0x1e | cpl=3 m+0x8000=000000001000 => #GP(0x10)
cli | cpl=3 => #GP(0)
hlt | cpl=3 => #GP(0)
rdmsr | cpl=3 rcx=0x10 => #GP(0)

# --- U773 harness: Windows reports a #SS as STATUS_ACCESS_VIOLATION with ExceptionInformation[0] = 3
# (address 0); a #PF has 0 (read), 1 (write) or 8 (execute) and the address, a #GP address -1
# (checked on this machine, Windows 10 19044, with our own snippets). vector_of mapped every
# access violation that was not a #GP to #PF (14). A non-canonical stack reference (RSP/RBP base,
# no FS/GS override) is #SS(0), any other #GP(0) (SDM Vol1 3.3.7.1; U51). Before U773 the five
# #SS lines differed (hw: fault #14, uc: fault #12).
mov rax, qword ptr [rsp+rcx] | rcx=0x00007FFFCFFEC0FC
mov rax, qword ptr [rsp+rcx] | rcx=0x8000000000000000
mov qword ptr [rsp+rcx], rax | rcx=0x8000000000000000
mov rax, qword ptr [rbp+rcx] | rbp=0x10 rcx=0x8000000000000000
add qword ptr [rsp+rcx+8], 1 | rcx=0x8000000000000000
mov rax, qword ptr fs:[rsp+rcx] | rcx=0x8000000000000000
mov rax, qword ptr ss:[rcx] | rcx=0x8000000000000000
mov rax, qword ptr [rcx] | rcx=0x8000000000000000
mov rax, qword ptr [rcx] | rcx=0x10

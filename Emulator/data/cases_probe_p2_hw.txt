# U1721 (G15 P2, notes\eread\G15_1J44_consolidation.md; sources S1sub_D V4/V5/V9, S1sub_C2 §3.2 D3, D5 §1.1
# RPL003 and §5.3 #22): debug fidelity, the part user mode can arrange WITHOUT debug registers. MEASUREMENT file,
# NOT a gate (not in test_runner.ps1); details and the guest results: notes\eread\P9_probe_snippets.md.
# Every line is tagged "# host state: measurement P2x ..." (a difference is counted per item, never as "differing").
# emu-alltest has no dr0..dr7 case tokens and MOV DR is #GP at CPL3 (cases_probe_p1_hw.txt P1c), so P2 (a) the
# instruction breakpoint after MOV SS / POP SS, (b) the data breakpoint on the MOV SS operand, (c) single-step with
# a not-yet-fired instruction breakpoint (Bn) and (e) DR6 after MOV SS are NOT here: they need debug registers
# (user mode only through the OS: SetThreadContext, which the harness lacks; else a kernel-mode probe). (d) DR6
# bit 11 is settled from the captures: CPUID.(7,0):ECX[24] (bus-lock detect) = 0 in the guest, bare-host P and
# bare-host E profiles, so DR6[11] stays fixed at 1. All of this is listed in the P9 note (host-data rule).
# What runs here: the single-step (RFLAGS.TF) side of the MOV SS / POP SS shadow. Each snippet sets TF with
# POPFQ (the trap comes after the instruction that follows POPF), then counts with INC R8 / INC ECX: the counter at
# the #DB shows after which instruction the trap was taken. Every snippet ends with spare instructions, so the
# trap RIP stays inside the snippet; the instruction after MOV SS / POP SS is always a plain INC / PUSH / NOP,
# never an instruction that enters the kernel (no INT n / INT3 / SYSCALL / fault in the shadow).
# Run (guest smoke / bare host, see the note for the v3 package lines):
#   emu-alltest --cases Emulator\data\cases_probe_p2_hw.txt --guest --hw-repeat 20 --hw-cpu all
#   emu-alltest --cases cases_probe_p2_hw.txt --host --cpuid cpuid_i5-13600k_bare_host.txt --hw-cpu P --hw-repeat 20
#   emu-alltest --cases cases_probe_p2_hw.txt --host --host-core E --cpuid cpuid_i5-13600k_bare_host_ecore.txt --hw-cpu E --hw-repeat 20
#
# --- control: TF set by POPFQ, the trap follows the next instruction (NOP): #DB with r8 = 0
pushfq; or qword ptr [rsp], 0x100; popfq; nop; inc r8; inc r8; inc r8; inc r8 | cpl=3  # host state: measurement P2-TF control
# --- MOV SS, r32 (8E D0) holds the trap over the next instruction: #DB with r8 = 1 (SDM Vol3A 7.8.3)
.byte 0x8c, 0xd0; pushfq; or qword ptr [rsp], 0x100; popfq; .byte 0x8e, 0xd0; inc r8; inc r8; inc r8; inc r8 | cpl=3  # host state: measurement P2-TF MOV SS reg
# --- MOV SS, m16 (8E 16 = mov ss, [rsi]): same
.byte 0x8c, 0xd0; mov word ptr [rsi], ax; pushfq; or qword ptr [rsp], 0x100; popfq; .byte 0x8e, 0x16; inc r8; inc r8; inc r8; inc r8 | cpl=3  # host state: measurement P2-TF MOV SS mem
# --- MOV SS; MOV SS: the second does not extend the shadow (SDM: only the first is guaranteed): r8 = 0 or 1
.byte 0x8c, 0xd0; pushfq; or qword ptr [rsp], 0x100; popfq; .byte 0x8e, 0xd0, 0x8e, 0xd0; inc r8; inc r8; inc r8; inc r8 | cpl=3  # host state: measurement P2-TF MOV SS twice
# --- MOV SS followed by a stack write (Intel's RPL003 workaround form): trap after the PUSH, r8 = 0, RSP - 8
.byte 0x8c, 0xd0; pushfq; or qword ptr [rsp], 0x100; popfq; .byte 0x8e, 0xd0; push rax; inc r8; inc r8; inc r8; inc r8 | cpl=3  # host state: measurement P2-TF MOV SS then PUSH
# --- POP SS (compatibility mode, CS = 23h via RETFQ; 32-bit code: mov eax, ss; push eax; pushfd; or dword [esp],
# 100h; popfd; pop ss; inc ecx x4): #DB with ecx = 1. Control: POP EAX instead of POP SS: ecx = 0.
.byte 0xe8, 0, 0, 0, 0, 0x5a, 0x48, 0x83, 0xc2, 0x0a, 0x6a, 0x23, 0x52, 0x48, 0xcb, 0x8c, 0xd0, 0x50, 0x9c, 0x81, 0x0c, 0x24, 0x00, 0x01, 0x00, 0x00, 0x9d, 0x17, 0x41, 0x41, 0x41, 0x41 | cpl=3  # host state: measurement P2-TF POP SS (compat)
.byte 0xe8, 0, 0, 0, 0, 0x5a, 0x48, 0x83, 0xc2, 0x0a, 0x6a, 0x23, 0x52, 0x48, 0xcb, 0x8c, 0xd0, 0x50, 0x9c, 0x81, 0x0c, 0x24, 0x00, 0x01, 0x00, 0x00, 0x9d, 0x58, 0x41, 0x41, 0x41, 0x41 | cpl=3  # host state: measurement P2-TF POP EAX control (compat)
# --- single-step priority against other events of the same instruction (TF = 1 at its start)
pushfq; or qword ptr [rsp], 0x100; popfq; int3; nop; nop; nop | cpl=3  # host state: measurement P2-TF over INT3 (#BP)
pushfq; or qword ptr [rsp], 0x100; popfq; ud2; nop; nop; nop | cpl=3  # host state: measurement P2-TF over UD2 (fault first)
pushfq; or qword ptr [rsp], 0x100; popfq; .byte 0xf1; inc r8; inc r8; inc r8 | cpl=3  # host state: measurement P2-TF over ICEBP
pushfq; or qword ptr [rsp], 0x100; popfq; mov eax, dword ptr [rsi+0x8000]; inc r8; inc r8; inc r8 | cpl=3  # host state: measurement P2-TF over a #PF load

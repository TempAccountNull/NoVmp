# --- U476 (backport 7653b44534): SGDT/SIDT store the whole base whatever the operand size. In
# 64-bit mode 66h does not change the 10-byte image (SDM Vol2B SGDT: "the operand size is fixed at
# 8+2 bytes"). The two images are compared on the CPU and the memory is cleared again, so the
# result does not depend on where the host or Unicorn keeps its GDT/IDT: rax = rcx = 0.

sidt [rsi]; .byte 0x66, 0x0f, 0x01, 0x4e, 0x10; mov rax, [rsi+2]; xor rax, [rsi+0x12]; movzx ecx, word ptr [rsi]; movzx edx, word ptr [rsi+0x10]; xor ecx, edx; xor edx, edx; mov [rsi], rdx; mov [rsi+8], rdx; mov [rsi+0x10], rdx; mov [rsi+0x18], rdx

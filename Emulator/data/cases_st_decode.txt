# ("RCX := RIP; (* Will contain address of next instruction *) RIP := IA32_LSTAR; R11 := RFLAGS; RFLAGS := RFLAGS AND

# (23H + 16 OR 3 = 33H), R12 := R12 - v (0: RCX held the address after SYSCALL); CLD; SYSCALL again -> h: R14 = the

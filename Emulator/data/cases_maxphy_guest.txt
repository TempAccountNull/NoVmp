# 2026-10-18). Moved out of cases_fixes2.txt (U593), which runs under both platform profiles. The guest profile
# (cpuid_i5-13600k.txt, the VMware guest capture) reports CPUID.80000008H:EAX = 302DH: MAXPHYADDR 45, so CR3 bits
# 44:12 are address bits and bits 63:45 reserved. SDM Vol2B MOV CR: "#GP(0) If an attempt is made to write a 1 to
# any reserved bit in CR3[63:MAXPHYADDR]"; Vol3A 5.5 (4-level paging, CR3 format). CPL0, Unicorn only:
#   emu-alltest --cases Emulator\data\cases_maxphy_guest.txt --guest --expect-only     (and --vm)
# The bare-host counterpart (MAXPHYADDR 46) is cases_maxphy_host.txt.

# --- bits 44:12 (and a bit-40 address): no fault, CR3 written
mov rax, 0x100000000000; mov cr3, rax => rax=0x100000000000
mov rax, 0x10000000000; mov cr3, rax => rax=0x10000000000
mov rax, 0x1FFFFFFFF000; mov cr3, rax => rax=0x1FFFFFFFF000
# --- bit 45 = MAXPHYADDR: the first reserved bit, #GP(0) (the cases_fixes2 line before U1652)

mov rax, 0x3FFFFFFFF000; mov cr3, rax => #GP(0) rax=0x3FFFFFFFF000
# --- bits 46 and 51: reserved
